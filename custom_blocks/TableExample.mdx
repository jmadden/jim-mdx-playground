import React, { useEffect, useState } from 'react';

export const ApiTable = ({ endpoint, columns, header }) => {
  const [rows, setRows] = useState([]);

  useEffect(() => {
    fetch(endpoint)
      .then(res => res.json())
      .then(data => setRows(data))
      .catch(err => console.error(err));
  }, [endpoint]);

  if (!rows.length) return <p className="text-center py-4">Loading…</p>;

  return (
    <div className="flex justify-center">
      <div className="rounded-2xl p-6 m-4 max-w-4xl shadow-lg border border-gray-200 dark:bg-gray-800 dark:border-gray-700">
        {/* Header */}
        {header && (
          <p className="text-2xl font-semibold mb-4 text-center">
            {header}
          </p>
        )}

        {/* Table */}
        <div className="overflow-x-auto">
          <table className="w-full table-auto border-collapse">
            <thead>
              <tr className="bg-gray-100 dark:bg-gray-700">
                {columns.map(col => (
                  <th
                    key={col}
                    className="px-4 py-2 text-left font-medium text-gray-700 dark:text-gray-300"
                  >
                    {col}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {rows.map((row, i) => (
                <tr
                  key={i}
                  className={i % 2 === 0 ? 'bg-white dark:bg-gray-900' : 'bg-gray-50 dark:bg-gray-800'}
                >
                  {columns.map(col => (
                    <td
                      key={col}
                      className="px-4 py-2 text-gray-800 dark:text-gray-200"
                    >
                      {typeof row[col] === 'object'
                        ? JSON.stringify(row[col])
                        : row[col]}
                    </td>
                  ))}
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};


<ApiTable
  header="User List"
  endpoint="https://jsonplaceholder.typicode.com/users"
  columns={['id', 'name', 'email', 'company']}
/>
