import React, { useEffect, useRef } from 'react';

export const SwaggerUIComponent = ({ specUrl }) => {
  const containerRef = useRef(null);

  useEffect(() => {
    // Helper to load external JavaScript
    const loadScript = (src) => {
      return new Promise((resolve, reject) => {
        if (document.querySelector(`script[src="${src}"]`)) {
          return resolve();
        }
        const script = document.createElement('script');
        script.src = src;
        script.onload = resolve;
        script.onerror = reject;
        document.body.appendChild(script);
      });
    };

    // Helper to load external CSS
    const loadCSS = (href) => {
      return new Promise((resolve, reject) => {
        if (document.querySelector(`link[href="${href}"]`)) {
          return resolve();
        }
        const link = document.createElement('link');
        link.rel = 'stylesheet';
        link.href = href;
        link.onload = resolve;
        link.onerror = reject;
        document.head.appendChild(link);
      });
    };

    // Load the Swagger UI assets from CDN
    Promise.all([
      loadCSS('https://unpkg.com/swagger-ui-dist/swagger-ui.css'),
      loadScript('https://unpkg.com/swagger-ui-dist/swagger-ui-bundle.js'),
      loadScript('https://unpkg.com/swagger-ui-dist/swagger-ui-standalone-preset.js')
    ])
      .then(() => {
        // Initialize Swagger UI once the assets are loaded, with branding hidden.
        if (window.SwaggerUIBundle && containerRef.current) {
          window.SwaggerUIBundle({
            url: specUrl,
            domNode: containerRef.current,
            presets: [
              window.SwaggerUIBundle.presets.apis,
              window.SwaggerUIStandalonePreset
            ],
            layout: "StandaloneLayout",
            hideTopbar: true, // This hides the Swagger UI top bar (branding)
          });
        }
      })
      .catch(error => {
        console.error("Error loading Swagger UI assets:", error);
      });
  }, [specUrl]);

  return (
    <div className="flex justify-center">
      <div 
        ref={containerRef}
        className="w-full max-w-4xl p-6 m-4 border border-gray-300 rounded-md shadow-md"
      >
        Loading API Documentation...
      </div>
    </div>
  );
};

<SwaggerUIComponent specUrl="https://developer.n-able.com/n-central/openapi/66e336d43f3d720013710c24" />
