import React, { useEffect, useRef } from 'react';

export const SwaggerUIComponent = ({ specUrl }) => {
  const containerRef = useRef(null);

  useEffect(() => {
    // Helper to load an external script
    const loadScript = (src) =>
      new Promise((resolve, reject) => {
        if (document.querySelector(`script[src="${src}"]`)) {
          return resolve();
        }
        const script = document.createElement('script');
        script.src = src;
        script.onload = resolve;
        script.onerror = reject;
        document.body.appendChild(script);
      });

    // Helper to load an external CSS file
    const loadCSS = (href) =>
      new Promise((resolve, reject) => {
        if (document.querySelector(`link[href="${href}"]`)) {
          return resolve();
        }
        const link = document.createElement('link');
        link.rel = 'stylesheet';
        link.href = href;
        link.onload = resolve;
        link.onerror = reject;
        document.head.appendChild(link);
      });

    Promise.all([
      loadCSS('https://unpkg.com/swagger-ui-dist/swagger-ui.css'),
      loadScript('https://unpkg.com/swagger-ui-dist/swagger-ui-bundle.js')
    ])
      .then(() => {
        if (window.SwaggerUIBundle && containerRef.current) {
          window.SwaggerUIBundle({
            url: specUrl,
            domNode: containerRef.current,
            presets: [window.SwaggerUIBundle.presets.apis],
            layout: "BaseLayout", // Minimal layout
            hideTopbar: true,
            docExpansion: "none" // Collapse operations by default
          });

          // Wait a moment for Swagger UI to render, then inject custom CSS.
          setTimeout(() => {
            const style = document.createElement('style');
            style.innerHTML = `
              /* Hide the header, info, operations, tags, and try-out buttons */
              .swagger-ui .topbar,
              .swagger-ui .info,
              .swagger-ui .opblock,
              .swagger-ui .opblock-tag,
              .swagger-ui .opblock-summary,
              .swagger-ui .schemes,
              .swagger-ui .btn,
              .swagger-ui .is-open {
                display: none !important;
              }
              /* Ensure the models (schemas) section is visible */
              .swagger-ui .models {
                display: block !important;
              }
            `;
            document.head.appendChild(style);
          }, 1000); // Adjust delay if needed
        }
      })
      .catch(error => {
        console.error("Error loading Swagger UI assets:", error);
      });
  }, [specUrl]);

  return (
    <div className="flex justify-center">
      <div 
        ref={containerRef}
        className="w-full max-w-4xl p-6 m-4 border border-gray-300 rounded-md shadow-md"
      >
        Loading API Documentation...
      </div>
    </div>
  );
};

<SwaggerUIComponent specUrl="https://developer.n-able.com/n-central/openapi/66e336d43f3d720013710c24" />
