export const MethodPill = ({ type, bg, color }) => {
  const defaultColors = {
    POST: { bg: '#ef4444', color: '#ffffff' },   // red-500
    GET: { bg: '#22c55e', color: '#ffffff' },    // green-500
    PATCH: { bg: '#facc15', color: '#000000' },  // yellow-400
    DELETE: { bg: '#f97316', color: '#ffffff' }, // orange-500
  };

  const style = {
    backgroundColor: bg || defaultColors[type]?.bg || '#e5e7eb', // fallback: gray-200
    color: color || defaultColors[type]?.color || '#000000',     // fallback: black
  };

  return (
    <span
      style={style}
      className="inline-block px-2 py-1 rounded-full text-xs font-bold font-mono"
    >
      {type}
    </span>
  );
};

<MethodPill type="POST" /> and <MethodPill type="GET" />
<MethodPill type="SYNC" bg="#8b5cf6" color="#fff" />
<MethodPill type="FETCH" bg="#0ea5e9" color="#000" />
