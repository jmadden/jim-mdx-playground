import React, { useState, useEffect } from 'react';

export const OpenAPISchemaViewer = () => {
  // A simple collapsible component.
  function Collapsible(props) {
    const [open, setOpen] = useState(true);
    return (
      <div className="mb-2 border border-gray-200 rounded">
        <div 
          className="cursor-pointer bg-gray-200 px-4 py-2 flex justify-between items-center"
          onClick={() => setOpen(!open)}
        >
          <span className="font-bold">{props.title}</span>
          <span>{open ? '-' : '+'}</span>
        </div>
        {open && <div className="p-4">{props.children}</div>}
      </div>
    );
  }

  // Nested component for rendering properties recursively.
  function PropertiesViewer(props) {
    const properties = props.properties;
    if (!properties) {
      return <p className="text-gray-500">No properties defined.</p>;
    }
    return (
      <ul className="ml-4 list-disc">
        {Object.entries(properties).map(function(item) {
          const propName = item[0];
          const details = item[1];
          return (
            <li key={propName} className="mb-1">
              <span className="font-semibold">{propName}</span>:{" "}
              <span className="text-blue-600">{details.type || "object"}</span>
              {details.description && (
                <span className="text-gray-600"> - {details.description}</span>
              )}
              {details.example !== undefined && (
                <div className="mt-1 p-2 bg-gray-100 border rounded text-sm">
                  Example: {JSON.stringify(details.example)}
                </div>
              )}
              {/* If nested properties exist, render them in a collapsible section */}
              {details.properties && (
                <Collapsible title="Nested Properties">
                  <PropertiesViewer properties={details.properties} />
                </Collapsible>
              )}
            </li>
          );
        })}
      </ul>
    );
  }

  // State for the schema, loading, and error.
  const [schemas, setSchemas] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  // Fetch the OpenAPI spec when the component mounts.
  useEffect(function() {
    fetch('https://developer.n-able.com/n-central/openapi/66e336d43f3d720013710c24')
      .then(function(response) {
        if (!response.ok) {
          throw new Error('HTTP error: ' + response.status);
        }
        return response.json();
      })
      .then(function(data) {
        // Extract schema data from components.schemas (OpenAPI 3.x) or definitions (OpenAPI 2.x)
        const schemaData = (data.components && data.components.schemas)
          ? data.components.schemas
          : data.definitions;
        setSchemas(schemaData);
      })
      .catch(function(err) {
        setError(err.message);
      })
      .finally(function() {
        setLoading(false);
      });
  }, []);

  if (loading) {
    return (
      <div className="p-4">
        Loading OpenAPI specification...
      </div>
    );
  }
  if (error) {
    return (
      <div className="p-4 text-red-600">
        Error loading specification: {error}
      </div>
    );
  }
  if (!schemas) {
    return (
      <div className="p-4">
        No schema data found in the OpenAPI specification.
      </div>
    );
  }

  return (
    <div className="space-y-6">
      <h1 className="text-3xl font-bold mb-4">API Schemas</h1>
      {Object.entries(schemas).map(function(item) {
        const schemaName = item[0];
        const schemaData = item[1];
        return (
          <Collapsible key={schemaName} title={schemaName}>
            {schemaData.description && (
              <p className="mb-2 text-gray-700">{schemaData.description}</p>
            )}
            {schemaData.example !== undefined && (
              <div className="mt-1 p-2 bg-gray-100 border rounded text-sm">
                Example: {JSON.stringify(schemaData.example)}
              </div>
            )}
            <PropertiesViewer properties={schemaData.properties} />
          </Collapsible>
        );
      })}
    </div>
  );
};

<OpenAPISchemaViewer />
