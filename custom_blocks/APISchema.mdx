import React, { useState, useEffect } from 'react';

export const OpenAPISchemaViewer = ({ specUrl, endpointPath, operation }) => {
  // Collapsible component with dark/light mode and global expand/contract control.
  const Collapsible = ({ title, children, globalExpand }) => {
    const [open, setOpen] = useState(false);
    useEffect(() => {
      if (typeof globalExpand === 'boolean') {
        setOpen(globalExpand);
      }
    }, [globalExpand]);
    return (
      <div className="mb-2 border border-gray-200 dark:border-gray-700 rounded">
        <div
          className="cursor-pointer bg-gray-200 dark:bg-gray-800 px-4 py-2 flex justify-between items-center"
          onClick={() => setOpen(!open)}
        >
          <span className="font-bold text-gray-900 dark:text-gray-100">{title}</span>
          <span className="text-gray-900 dark:text-gray-100">{open ? '-' : '+'}</span>
        </div>
        {open && <div className="p-4 bg-white dark:bg-gray-900">{children}</div>}
      </div>
    );
  };

  // Recursively render properties.
  const PropertiesViewer = ({ properties, globalExpand }) => {
    if (!properties)
      return <p className="text-gray-500 dark:text-gray-400">No properties defined.</p>;
    return (
      <ul className="ml-4 list-disc">
        {Object.entries(properties).map(([propName, details]) => (
          <li key={propName} className="mb-1">
            <span className="font-semibold text-gray-900 dark:text-gray-100">{propName}</span>:{" "}
            <span className="text-blue-600 dark:text-blue-400">{details.type || "object"}</span>
            {details.description && (
              <span className="text-gray-600 dark:text-gray-300"> - {details.description}</span>
            )}
            {details.example !== undefined && (
              <div className="mt-1">
                <span className="font-semibold text-gray-900 dark:text-gray-100">Example:</span>
                <pre
                  className="p-2 bg-gray-100 dark:bg-gray-700 rounded text-sm whitespace-pre-wrap"
                  style={{ paddingLeft: "5px" }}
                >
                  {typeof details.example === 'object'
                    ? JSON.stringify(details.example, null, 2)
                    : details.example}
                </pre>
              </div>
            )}
            {details.properties && (
              <Collapsible title="Nested Properties" globalExpand={globalExpand}>
                <PropertiesViewer properties={details.properties} globalExpand={globalExpand} />
              </Collapsible>
            )}
          </li>
        ))}
      </ul>
    );
  };

  // Local state for schema data, loading, error, and global expand/contract control.
  const [schemas, setSchemas] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [globalExpand, setGlobalExpand] = useState(false);

  useEffect(() => {
    fetch(specUrl)
      .then((response) => {
        if (!response.ok) throw new Error(`HTTP error: ${response.status}`);
        return response.json();
      })
      .then((data) => {
        let schemaData;
        if (endpointPath && operation) {
          // Look for the specified endpoint and method.
          const endpointSpec = data.paths?.[endpointPath]?.[operation.toLowerCase()];
          // Extract the requestBody schema for application/json (adjust if needed).
          const reqSchema = endpointSpec?.requestBody?.content?.['application/json']?.schema;
          if (reqSchema) {
            // Wrap it in an object so that the viewer can treat it like other schema definitions.
            schemaData = { [`${operation.toUpperCase()} ${endpointPath}`]: reqSchema };
          }
        } else {
          // Fallback to full spec's components.schemas (or definitions if OAS 2.0).
          schemaData = data.components?.schemas ?? data.definitions;
        }
        setSchemas(schemaData);
      })
      .catch((err) => setError(err.message))
      .finally(() => setLoading(false));
  }, [specUrl, endpointPath, operation]);

  if (loading)
    return <div className="p-4 text-gray-900 dark:text-gray-100">Loading OpenAPI specification...</div>;
  if (error)
    return <div className="p-4 text-red-600">Error loading specification: {error}</div>;
  if (!schemas)
    return <div className="p-4 text-gray-900 dark:text-gray-100">No schema data found in the OpenAPI specification.</div>;

  return (
    <div className="space-y-6">
      {/* Global control buttons */}
      <div className="mb-4 flex space-x-4">
        <button
          className="px-4 py-2 bg-blue-500 dark:bg-blue-600 text-white rounded"
          onClick={() => setGlobalExpand(true)}
        >
          Expand All
        </button>
        <button
          className="px-4 py-2 bg-blue-500 dark:bg-blue-600 text-white rounded"
          onClick={() => setGlobalExpand(false)}
        >
          Contract All
        </button>
      </div>
      <h1 className="text-3xl font-bold mb-4 text-gray-900 dark:text-gray-100">API Schemas</h1>
      {Object.entries(schemas).map(([schemaName, schemaData]) => (
        <Collapsible key={schemaName} title={schemaName} globalExpand={globalExpand}>
          {schemaData.description && (
            <p className="mb-2 text-gray-700 dark:text-gray-300">{schemaData.description}</p>
          )}
          {schemaData.example !== undefined && (
            <div className="mt-1">
              <span className="font-semibold text-gray-900 dark:text-gray-100">Example:</span>
              <pre
                className="p-2 bg-gray-100 dark:bg-gray-700 rounded text-sm whitespace-pre-wrap"
                style={{ paddingLeft: "5px" }}
              >
                {typeof schemaData.example === 'object'
                  ? JSON.stringify(schemaData.example, null, 2)
                  : schemaData.example}
              </pre>
            </div>
          )}
          <PropertiesViewer properties={schemaData.properties} globalExpand={globalExpand} />
        </Collapsible>
      ))}
    </div>
  );
};

<OpenAPISchemaViewer 
  specUrl="https://mydomain.com/project/openapi" 
  endpointPath="/api/devices/{deviceId}/assets/lifecycle-inf" 
  operation="put" 
/>
