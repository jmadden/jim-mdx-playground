import React, { useState, useEffect } from 'react';

export const OpenAPISchemaViewer = ({ specUrl }) => {
  // Collapsible component that listens for a global expand/contract command.
  const Collapsible = ({ title, children, globalExpand }) => {
    const [open, setOpen] = useState(false);
    useEffect(() => {
      if (typeof globalExpand === 'boolean') {
        setOpen(globalExpand);
      }
    }, [globalExpand]);
    return (
      <div className="mb-2 border border-gray-200 rounded">
        <div
          className="cursor-pointer bg-gray-200 px-4 py-2 flex justify-between items-center"
          onClick={() => setOpen(!open)}
        >
          <span className="font-bold">{title}</span>
          <span>{open ? '-' : '+'}</span>
        </div>
        {open && <div className="p-4">{children}</div>}
      </div>
    );
  };

  // Recursively render properties with pretty examples and nested collapsible sections.
  const PropertiesViewer = ({ properties, globalExpand }) => {
    if (!properties) return <p className="text-gray-500">No properties defined.</p>;
    return (
      <ul className="ml-4 list-disc">
        {Object.entries(properties).map(([propName, details]) => (
          <li key={propName} className="mb-1">
            <span className="font-semibold">{propName}</span>:{" "}
            <span className="text-blue-600">{details.type || "object"}</span>
            {details.description && (
              <span className="text-gray-600"> - {details.description}</span>
            )}
            {details.example !== undefined && (
              <div className="mt-1">
                <span className="font-semibold">Example:</span>
                <pre 
                  className="p-2 bg-gray-100 rounded text-sm whitespace-pre-wrap"
                  style={{ paddingLeft: "5px" }}
                >
                  {typeof details.example === 'object'
                    ? JSON.stringify(details.example, null, 2)
                    : details.example}
                </pre>
              </div>
            )}
            {details.properties && (
              <Collapsible title="Nested Properties" globalExpand={globalExpand}>
                <PropertiesViewer properties={details.properties} globalExpand={globalExpand} />
              </Collapsible>
            )}
          </li>
        ))}
      </ul>
    );
  };

  // Local state for the fetched schema, loading, error, and global expand/contract control.
  const [schemas, setSchemas] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [globalExpand, setGlobalExpand] = useState(false);

  useEffect(() => {
    fetch(specUrl)
      .then((response) => {
        if (!response.ok) throw new Error(`HTTP error: ${response.status}`);
        return response.json();
      })
      .then((data) => {
        const schemaData = data.components?.schemas ?? data.definitions;
        setSchemas(schemaData);
      })
      .catch((err) => setError(err.message))
      .finally(() => setLoading(false));
  }, [specUrl]);

  if (loading) return <div className="p-4">Loading OpenAPI specification...</div>;
  if (error) return <div className="p-4 text-red-600">Error loading specification: {error}</div>;
  if (!schemas) return <div className="p-4">No schema data found in the OpenAPI specification.</div>;

  return (
    <div className="space-y-6">
      {/* Global control buttons */}
      <div className="mb-4 flex space-x-4">
        <button
          className="px-4 py-2 bg-blue-500 text-white rounded"
          onClick={() => setGlobalExpand(true)}
        >
          Expand All
        </button>
        <button
          className="px-4 py-2 bg-blue-500 text-white rounded"
          onClick={() => setGlobalExpand(false)}
        >
          Collaps All
        </button>
      </div>
      <h1 className="text-3xl font-bold mb-4">API Schemas</h1>
      {Object.entries(schemas).map(([schemaName, schemaData]) => (
        <Collapsible key={schemaName} title={schemaName} globalExpand={globalExpand}>
          {schemaData.description && (
            <p className="mb-2 text-gray-700">{schemaData.description}</p>
          )}
          {schemaData.example !== undefined && (
            <div className="mt-1">
              <span className="font-semibold">Example:</span>
              <pre 
                className="p-2 bg-gray-100 rounded text-sm whitespace-pre-wrap"
                style={{ paddingLeft: "5px" }}
              >
                {typeof schemaData.example === 'object'
                  ? JSON.stringify(schemaData.example, null, 2)
                  : schemaData.example}
              </pre>
            </div>
          )}
          <PropertiesViewer properties={schemaData.properties} globalExpand={globalExpand} />
        </Collapsible>
      ))}
    </div>
  );
};

<OpenAPISchemaViewer specUrl="https://mydomain.com/project/openapi" />
