import React, { useState } from 'react';

export const InteractiveExample = ({ header, details }) => {
  const [showDetails, setShowDetails] = useState(false);

  return (
    <div className="flex justify-center">
      <div className="rounded-md p-6 m-4 max-w-lg shadow-md border border-gray-300">
        <p className="text-lg font-bold">{header}</p>
        <p className="mt-2">
          <i className="fa-solid fa-hand-pointer mr-2" />
          Click the button to {showDetails ? 'hide' : 'show'} additional details.
        </p>
        <button
          onClick={() => setShowDetails(!showDetails)}
          className="mt-4 px-4 py-2 bg-blue-500 text-white rounded"
        >
          {showDetails ? 'Hide Details' : 'Show Details'}
        </button>
        {showDetails && (
          <div className="mt-4 p-4 bg-gray-100 border border-gray-200 rounded">
            {details}
          </div>
        )}
      </div>
    </div>
  );
};
